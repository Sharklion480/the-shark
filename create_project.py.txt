created_folders = 0
created_files = 0
skipped_files = 0

# انشاء كل المجلدات اولا
all_folders = list(set([str(Path(file).parent) for file in project_structure]))
all_folders.sort()

print("\n📁 جاري انشاء المجلدات...")
for folder in all_folders:
    Path(folder).mkdir(parents=True, exist_ok=True)
    created_folders += 1
    print(f"✅ {folder}")

print(f"\n✅ تم انشاء {created_folders} مجلد بنجاح")
print("\n📄 جاري انشاء الملفات...")

# انشاء كل الملفات
for file_path in project_structure:
    path = Path(file_path)
    
    if path.exists():
        skipped_files +=1
        continue
    
    # انشاء الملف فارغ
    path.touch()
    
    # اضافة __init__.py افتراضي
    if path.name == "__init__.py":
        path.write_text(f'"""\n{path.parent}\n"""\n', encoding='utf-8')
    
    created_files +=1
    print(f"✅ {file_path}")

print("=" * 70)
print("✅ 🎉 تم اكتمال انشاء هيكل المشروع بالكامل!")
print("=" * 70)
